```sql
SELECT stock_code, date, total_win_ratio_score, volume_type, updated_at
FROM b_daily_chance
WHERE date = CURDATE()
ORDER BY updated_at DESC
LIMIT 10;

-- 统计今天更新的股票数（应该是59支）
SELECT COUNT(DISTINCT stock_code) as count
FROM b_daily_chance
WHERE date = CURDATE() AND DATE(updated_at) = CURDATE();
```

